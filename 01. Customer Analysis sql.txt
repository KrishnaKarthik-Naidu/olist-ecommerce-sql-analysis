 =====================================================
  Olist E-Commerce Sales
  01_customer_analysis
 =====================================================


1. Total Customers

SELECT
    COUNT(*) AS total_customer_records
FROM customers;





2. Total Unique Customers

SELECT
    COUNT(DISTINCT customer_unique_id) AS unique_customers
FROM customers;





3. Customer Distribution by State

SELECT
    customer_state,
    COUNT(DISTINCT customer_unique_id) AS unique_customers
FROM customers
GROUP BY customer_state
ORDER BY unique_customers DESC;






4. Repeated Customers

SELECT
    c.customer_unique_id,
    COUNT(o.order_id) AS order_count
FROM customers c
JOIN orders o
    ON c.customer_id = o.customer_id
GROUP BY c.customer_unique_id
HAVING COUNT(o.order_id) > 1
ORDER BY order_count DESC;





5. Top 10 Customers by Total Spending

WITH order_ids AS (
    SELECT
        c.customer_unique_id,
        o.order_id
    FROM customers c
    JOIN orders o
        ON c.customer_id = o.customer_id
),
order_totals AS (
    SELECT
        oi.customer_unique_id,
        oi.order_id,
        SUM(p.payment_value) AS order_total
    FROM payments p
    JOIN order_ids oi
        ON oi.order_id = p.order_id
    GROUP BY
        oi.customer_unique_id,
        oi.order_id
)
SELECT
    customer_unique_id,
    SUM(order_total) AS total_spent
FROM order_totals
GROUP BY customer_unique_id
ORDER BY total_spent DESC
LIMIT 10;
