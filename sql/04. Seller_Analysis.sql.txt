
 1. Total Sellers

SELECT
    COUNT(*) AS total_sellers
FROM sellers;





2. Total Sales by Seller

SELECT
    seller_id,
    SUM(price) AS total_sales
FROM order_items
GROUP BY seller_id
ORDER BY total_sales DESC;





3. Top 10 Sellers by Sales

SELECT
    seller_id,
    SUM(price) AS total_sales
FROM order_items
GROUP BY seller_id
ORDER BY total_sales DESC
LIMIT 10;





4. Orders Handled by Each Seller

SELECT
    seller_id,
    COUNT(DISTINCT order_id) AS total_orders
FROM order_items
GROUP BY seller_id
ORDER BY total_orders DESC;





5. Average Sales per Order by Seller

SELECT
    seller_id,
    SUM(price) / COUNT(DISTINCT order_id) AS average_sales_per_order
FROM order_items
GROUP BY seller_id
ORDER BY average_sales_per_order DESC;






6. Top 10 Sellers by Number of Orders

SELECT
    seller_id,
    COUNT(DISTINCT order_id) AS total_orders
FROM order_items
GROUP BY seller_id
ORDER BY total_orders DESC
LIMIT 10;





7. Sellers with More Than 100 Orders

SELECT
    seller_id,
    COUNT(DISTINCT order_id) AS total_orders
FROM order_items
GROUP BY seller_id
HAVING COUNT(DISTINCT order_id) > 100
ORDER BY total_orders DESC;
